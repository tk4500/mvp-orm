schema.nexus.ts — Enterprise Order Ledger
TypeScript / ACID
import { model, field, relation, erpLedger } from '@nexus/orm';

// Multi-tenant enterprise ledger model with automated audit trails
@erpLedger({ currency: 'USD', isolation: 'SERIALIZABLE' })
export class EnterpriseOrder extends Model {
@field({ type: UUID, primary: true, default: sql`gen_random_uuid()` })
id!: string;

@field({ type: VarChar(128), index: true, auditLog: true })
orderReference!: string;

@relation({ model: Customer, onDelete: 'RESTRICT' })
customer!: Customer;

@field({ type: Decimal(14, 4), precision: 4, immutable: true })
totalAmount!: number;
}
