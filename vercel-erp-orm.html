schema.nexus.ts — Enterprise Order & Inventory Model
import { model, field, relation, erpModule } @nexus/orm;

// Define multi-tenant ERP Ledger with automated ACID constraints
@erpModule({ name: "SupplyChain & Ledger" })
export class EnterpriseOrder extends Model {
  @field({ type: UUID, primary: true, default: sql`gen_random_uuid()` })
  id!: string;

  @field({ type: VarChar(128), index: true })
  orderNumber!: string;

  @relation({ model: Customer, onDelete: "RESTRICT" })
  customer!: Customer;

  @field({ type: Decimal(12, 4), precision: 4, auditLog: true })
  totalAmount!: number;
}
